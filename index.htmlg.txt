            >

                <option value="__NEW__">
                    ＋ নতুন পণ্য লিখুন
                </option>

            </select>

        </td>

        <td>

            <select
                class="sizeSelect"

            >

                <option value="">
                    সাইজ নির্বাচন
                </option>

                <option value="__NEW_SIZE__">
                    ＋ নতুন সাইজ লিখুন

    tbody.appendChild(tr);

    updateEmpty();
}


/* ==================================================
   PRODUCT OPTIONS
================================================== */

function productOptions(){

    let html="";

    Object.keys(products).forEach(product=>{

        html+=`
            <option value="${escapeHTML(product)}">
                ${escapeHTML(product)}
            </option>
        `;

    });

    return html;
}


/* ==================================================
   PRODUCT CHANGE
================================================== */

function productChange(select){

    const row=
        select.closest("tr");

    const size=
        row.querySelector(".sizeSelect");


    if(select.value==="__NEW__"){

        const name=
            prompt("নতুন পণ্যের নাম লিখুন:");

        if(!name || !name.trim()){

            select.value="";
            return;
        }

        const productName=
            name.trim();

        if(!products[productName]){
            products[productName]=[];
        }


        const input=
            document.createElement("input");

        input.className=
            "custom-input productCustom";

        input.value=
            productName;

        input.placeholder=
            "পণ্যের নাম লিখুন";

        input.oninput=
            updatePreview;

        select.replaceWith(input);

        makeSizeList(
            size,
            productName
        );

        updatePreview();

        return;
    }


    makeSizeList(
        size,
        select.value
    );

    updatePreview();
}

   SIZE LIST
================================================== */

function makeSizeList(
    select,
    product
){

    select.innerHTML=`
        <option value="">
            সাইজ নির্বাচন
        </option>
    `;

    const sizes=
        products[product] || [];


    sizes.forEach(size=>{

        const option=
            document.createElement("option");

        option.value=size;
        option.textContent=size;

        select.appendChild(option);

    });


    const custom=
        document.createElement("option");

    custom.value=
        "__NEW_SIZE__";

    custom.textContent=
        "＋ নতুন সাইজ লিখুন";

    select.appendChild(custom);
}


/* ==================================================
   SIZE CHANGE
================================================== */

function sizeChange(select){

    if(
        select.value!=="__NEW_SIZE__"
    ){

        updatePreview();
        return;
    }


    const value=
        prompt("সাইজ লিখুন:");

    if(!value || !value.trim()){

        select.value="";
        return;
    }


    const input=
        document.createElement("input");

    input.className=
        "custom-input sizeCustom";

    input.value=
        value.trim();

    input.placeholder=
        "সাইজ লিখুন";

    input.oninput=
        updatePreview;

    select.replaceWith(input);

    updatePreview();
}

/* ==================================================
   UPDATE PREVIEW
================================================== */

function updatePreview(){

    const left=
        document.getElementById(
            "leftPreview"
        );

    const right=
        document.getElementById(
            "rightPreview"
        );

    left.innerHTML="";
    right.innerHTML="";


    const rows=
        document.querySelectorAll(
            "#productBody tr"
        );


    rows.forEach(
        (row,index)=>{

            const no=
                index+1;

            const product=
                getProduct(row);

            const size=
                getSize(row);

            const qty=
                row.querySelector(
                    ".qty-input"
                )?.value.trim() || "";


            const tr=
                document.createElement("tr");

            tr.innerHTML=`

                <td class="pno">
                    ${no}
                </td>

                <td class="pname">
                    ${escapeHTML(product)}
                </td>

                <td class="psize">
                    ${escapeHTML(size)}
                </td>

                <td class="pqty">
                    ${escapeHTML(qty)}
                </td>

            `;


            /*
              LEFT = 1-25
              RIGHT = 26-50
            */

            if(no<=25){

                left.appendChild(tr);

            }else{

                right.appendChild(tr);

            }

        }
    );


    document.getElementById(
        "previewName"
    ).textContent=
        document.getElementById(
            "customerName"
        ).value;


    document.getElementById(
        "previewPhone"
    ).textContent=
        document.getElementById(
            "customerPhone"
        ).value;


    document.getElementById(
        "previewAddress"
    ).textContent=
        document.getElementById(
            "customerAddress"
        ).value;
}

/* ==================================================
   ESCAPE HTML
================================================== */

function escapeHTML(value){

    return String(value || "")
        .replace(/&/g,"&amp;")
        .replace(/</g,"&lt;")
        .replace(/>/g,"&gt;")
        .replace(/"/g,"&quot;")
        .replace(/'/g,"&#039;");
}
